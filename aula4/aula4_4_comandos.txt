FASE 1 - IDENTIFICACAO
================================================================================

--- Windows ---
netstat -ano | findstr "LISTENING"
schtasks /query /fo LIST /v | findstr /i "update"
reg query "HKLM\Software\Microsoft\Windows\CurrentVersion\Run"
Get-NetFirewallRule | Where-Object { (Get-NetFirewallPortFilter -AssociatedNetFirewallRule $_).LocalPort -eq 2222 }

--- Linux (sudo su -) ---
ps aux | grep -E "\.(sh|py|pl|rb)$|backdoor|reverse|nc |netcat" | grep -v grep
ss -tlnp
crontab -l
cat /etc/cron.d/nginx-backdoor 2>/dev/null
systemctl list-units --type=service | grep nginx

================================================================================
FASE 2 - REMOCAO
================================================================================

--- Windows ---
$pid = (Get-NetTCPConnection -LocalPort 2222).OwningProcess; if ($pid) { Stop-Process -Id $pid -Force }
schtasks /delete /tn "WindowsUpdate" /f
reg delete "HKLM\Software\Microsoft\Windows\CurrentVersion\Run" /v "Windows Service" /f
mkdir C:\Quarentena; move "C:\ProgramData\Microsoft\winupdate.bat" C:\Quarentena\
Remove-NetFirewallRule -DisplayName "Allow_SSH_Backdoor"

--- Linux (sudo su -) ---
pkill -f "nginx-backdoor"
pkill -f "nc -l -p 4444"
crontab -l | grep -v "nginx-backdoor" | crontab -
rm /etc/cron.d/nginx-backdoor 2>/dev/null
systemctl stop nginx-backup 2>/dev/null
systemctl disable nginx-backup 2>/dev/null
rm /etc/systemd/system/nginx-backup.service 2>/dev/null
systemctl daemon-reload
mkdir -p /quarentena
mv /tmp/.nginx-backdoor /quarentena/ 2>/dev/null
mv /tmp/.backdoor.log /quarentena/ 2>/dev/null

================================================================================
FASE 4 - VALIDACAO DE INTEGRIDADE
================================================================================

--- Windows ---
Get-ChildItem "C:\DB_Data\" -File | ForEach-Object {
    $hash = (Get-FileHash -Path $_.FullName -Algorithm SHA256).Hash
    "$hash  $($_.Name)"
} | Out-File "C:\DB_Data\manifesto_restaurado.sha256"
Compare-Object (Get-Content "C:\Backup\manifesto_backup.sha256") (Get-Content "C:\DB_Data\manifesto_restaurado.sha256")

--- Linux ---
sha256sum /data/db/* > /data/db/manifesto_restaurado.sha256
grep -v "^#" /backup/manifesto_backup.sha256 | sort > /tmp/backup_sorted.txt
grep -v "^#" /data/db/manifesto_restaurado.sha256 | sort > /tmp/restaurado_sorted.txt
diff /tmp/backup_sorted.txt /tmp/restaurado_sorted.txt
