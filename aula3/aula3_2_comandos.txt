================================================================================
METODO 1: FIREWALL LOCAL
================================================================================

# Windows - Bloquear todo trafego
netsh advfirewall set allprofiles firewallpolicy blockinbound,blockoutbound

# Windows - Verificar politica aplicada
netsh advfirewall show allprofiles

# Windows - Bloqueio granular (IP especifico)
netsh advfirewall firewall add rule name="Block_C2" dir=out remoteip=203.0.113.45 protocol=tcp action=block

# Windows - Bloqueio granular (portas de movimento lateral)
netsh advfirewall firewall add rule name="Block_SMB_Out" dir=out remoteport=445 protocol=tcp action=block
netsh advfirewall firewall add rule name="Block_RDP_Out" dir=out remoteport=3389 protocol=tcp action=block

# Windows - Reverter firewall
netsh advfirewall set allprofiles firewallpolicy blockinbound,allowoutbound

# Linux - Bloquear todo trafego (iptables)
sudo iptables -P INPUT DROP
sudo iptables -P OUTPUT DROP
sudo iptables -P FORWARD DROP
sudo iptables -A INPUT -i lo -j ACCEPT
sudo iptables -A OUTPUT -o lo -j ACCEPT

# Linux - Verificar regras
sudo iptables -L -v

# Linux - Reverter iptables
sudo iptables -P INPUT ACCEPT
sudo iptables -P OUTPUT ACCEPT
sudo iptables -P FORWARD ACCEPT
sudo iptables -F

# Linux - Bloquear com ufw
sudo ufw enable
sudo ufw default deny incoming
sudo ufw default deny outgoing

# Linux - Reverter ufw
sudo ufw disable

================================================================================
METODO 2: DESABILITAR ADAPTADOR DE REDE
================================================================================

# Windows - Listar adaptadores
Get-NetAdapter

# Windows - Desabilitar TODOS os adaptadores
Get-NetAdapter | Disable-NetAdapter -Confirm:$false

# Windows - Reabilitar todos os adaptadores
Get-NetAdapter | Enable-NetAdapter -Confirm:$false

# Windows - Desabilitar um adaptador especifico
netsh interface set interface "Ethernet 2" admin=disable

# Windows - Reabilitar um adaptador especifico
netsh interface set interface "Ethernet 2" admin=enable

# Linux - Listar interfaces
ip link show

# Linux - Desabilitar interface
sudo ip link set enp0s8 down

# Linux - Reabilitar interface
sudo ip link set enp0s8 up

# Linux - Via NetworkManager
sudo nmcli device disconnect enp0s8
sudo nmcli device connect enp0s8
