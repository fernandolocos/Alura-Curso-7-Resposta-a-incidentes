FASE 2 - REMOCAO WINDOWS
================================================================================

--- 1. Remover conta backdoor primeiro ---

net localgroup Administrators svc_monitor /delete
net user svc_monitor /delete
net user svc_monitor

--- 2. Remover tarefa agendada ---

Disable-ScheduledTask -TaskName "MicrosoftEdgeUpdateTask"
Unregister-ScheduledTask -TaskName "MicrosoftEdgeUpdateTask" -Confirm:$false
Get-ScheduledTask -TaskName "MicrosoftEdgeUpdateTask"

--- 3. Mover binario para quarentena ---

move "C:\ProgramData\Microsoft\monitor_agent.exe" C:\Quarentena\

--- Windows ---

Get-ChildItem "C:\DB_Data\" -File | Where-Object { $_.Name -ne "manifesto_restaurado.sha256" } | ForEach-Object {
    $hash = (Get-FileHash -Path $_.FullName -Algorithm SHA256).Hash
    "$hash  $($_.Name)"
} | Out-File "C:\DB_Data\manifesto_restaurado.sha256"
